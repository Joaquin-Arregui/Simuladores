# Modelo de referencia 27 - parejas con una cadena de respuestas (D7, 3.4).
# Tres parejas 1:1, con los clientes llegando cada cien segundos. Cada servicio
# avisa a SU cliente, el cliente le pide en respuesta al aviso, y el servicio
# responde a la peticion; el cliente recoge las respuestas con una recepcion
# multiinstancia. Mientras el servicio 3 no ha recibido la peticion de su
# cliente, todavia podria anotar a alguien y responderle: la recepcion del
# cliente 1 tiene que saber que ese alguien solo puede ser el cliente 3, y
# eso pasa por ver que la peticion es a su vez una respuesta, que va a quien
# avisaron. Si no se sigue la cadena y se da por hecho que podria ser
# cualquiera, el cliente 1, que tiene su respuesta en t=43, se queda esperando
# hasta t=206, cuando el servicio 3 recibe por fin la peticion de su cliente.
Element: [type=bpmn:Collaboration, id_bpmn='Colaboracion', name='Parejas con cadena de respuestas', instances=3]
Element: [type=bpmn:Participant, id_bpmn='Pool_Cliente', name='Cliente', frequency=100, userWithoutRole=['Carla'], infiniteResources='true', containedElements=['Inicio_C', 'Esperar_Aviso', 'Enviar_Peticion', 'Esperar_Respuesta', 'Fin_C', 'c1', 'c2', 'c3', 'c4']]
Element: [type=bpmn:Participant, id_bpmn='Pool_Servicio', name='Servicio', frequency=0, userWithoutRole=['Sara'], infiniteResources='true', containedElements=['Inicio_S', 'Avisar', 'Recibir_Peticion', 'Atender', 'Responder', 'Fin_S', 's1', 's2', 's3', 's4', 's5']]
Element: [type=bpmn:StartEvent, id_bpmn='Inicio_C', name='Llega el cliente', subTask='Esperar_Aviso']
Element: [type=bpmn:ReceiveTask, id_bpmn='Esperar_Aviso', name='Esperar el aviso', userTask='Carla', numberOfExecutions=1, minimumTime=1, maximumTime=2, subTask='Enviar_Peticion']
Element: [type=bpmn:SendTask, id_bpmn='Enviar_Peticion', name='Pedir', userTask='Carla', numberOfExecutions=1, minimumTime=2, maximumTime=5, subTask='Esperar_Respuesta']
Element: [type=bpmn:ReceiveTask, id_bpmn='Esperar_Respuesta', name='Recoger las respuestas', userTask='Carla', numberOfExecutions=1, multiInstanceType='false', minimumTime=1, maximumTime=2, subTask='Fin_C']
Element: [type=bpmn:EndEvent, id_bpmn='Fin_C', name='Se va el cliente', subTask='']
Element: [type=bpmn:StartEvent, id_bpmn='Inicio_S', name='Arranca el servicio', subTask='Avisar']
Element: [type=bpmn:SendTask, id_bpmn='Avisar', name='Avisar a su cliente', userTask='Sara', numberOfExecutions=1, minimumTime=1, maximumTime=2, subTask='Recibir_Peticion']
Element: [type=bpmn:ReceiveTask, id_bpmn='Recibir_Peticion', name='Recibir la peticion', userTask='Sara', numberOfExecutions=1, minimumTime=1, maximumTime=2, subTask='Atender']
Element: [type=bpmn:Task, id_bpmn='Atender', name='Atender la peticion', userTask='Sara', numberOfExecutions=1, minimumTime=20, maximumTime=40, subTask='Responder']
Element: [type=bpmn:SendTask, id_bpmn='Responder', name='Responder', userTask='Sara', numberOfExecutions=1, minimumTime=1, maximumTime=2, subTask='Fin_S']
Element: [type=bpmn:EndEvent, id_bpmn='Fin_S', name='Fin del servicio', subTask='']
Element: [type=bpmn:SequenceFlow, id_bpmn='c1', name='', superElement='Inicio_C', subElement='Esperar_Aviso']
Element: [type=bpmn:SequenceFlow, id_bpmn='c2', name='', superElement='Esperar_Aviso', subElement='Enviar_Peticion']
Element: [type=bpmn:SequenceFlow, id_bpmn='c3', name='', superElement='Enviar_Peticion', subElement='Esperar_Respuesta']
Element: [type=bpmn:SequenceFlow, id_bpmn='c4', name='', superElement='Esperar_Respuesta', subElement='Fin_C']
Element: [type=bpmn:SequenceFlow, id_bpmn='s1', name='', superElement='Inicio_S', subElement='Avisar']
Element: [type=bpmn:SequenceFlow, id_bpmn='s2', name='', superElement='Avisar', subElement='Recibir_Peticion']
Element: [type=bpmn:SequenceFlow, id_bpmn='s3', name='', superElement='Recibir_Peticion', subElement='Atender']
Element: [type=bpmn:SequenceFlow, id_bpmn='s4', name='', superElement='Atender', subElement='Responder']
Element: [type=bpmn:SequenceFlow, id_bpmn='s5', name='', superElement='Responder', subElement='Fin_S']
Element: [type=bpmn:MessageFlow, id_bpmn='Mensaje_Aviso', name='Aviso', superElement='Avisar', subElement='Esperar_Aviso']
